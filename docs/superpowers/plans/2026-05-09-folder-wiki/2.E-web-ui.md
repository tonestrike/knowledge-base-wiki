```ts
// apps/web/src/lib/use-event-stream.ts
import { useEffect, useState } from 'react';

export function useEventStream<T>(
  url: string | null,
  parse: (raw: unknown) => T,
): { events: T[]; done: boolean; error: string | null } {
  const [events, setEvents] = useState<T[]>([]);
  const [done, setDone] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (!url) return;
    setEvents([]); setDone(false); setError(null);
    const ac = new AbortController();
    (async () => {
      try {
        const res = await fetch(url, { signal: ac.signal });
        if (!res.body) throw new Error('no response body');
        const reader = res.body.getReader();
        const decoder = new TextDecoder();
        let buf = '';
        while (true) {
          const { done, value } = await reader.read();
          if (done) break;
          buf += decoder.decode(value, { stream: true });
          let idx;
          while ((idx = buf.indexOf('\n\n')) !== -1) {
            const evt = buf.slice(0, idx).replace(/^data:\s*/m, '').trim();
            buf = buf.slice(idx + 2);
            if (!evt) continue;
            const parsed = parse(JSON.parse(evt));
            setEvents((prev) => [...prev, parsed]);
          }
        }
        setDone(true);
      } catch (e) {
        if ((e as Error).name !== 'AbortError') setError(String(e));
      }
    })();
    return () => ac.abort();
  }, [url, parse]);

  return { events, done, error };
}
```

```ts
// apps/web/src/components/citation/byte-range-to-bbox.ts
export interface OutlineNode { kind: 'heading' | 'table' | 'figure'; level: number; title: string; byteRange: { start: number; end: number }; page?: number }

export function pageForByteRange(outline: OutlineNode[], byteRange: { start: number; end: number }): number {
  // Pick the latest outline node whose start ≤ byteRange.start AND has a `page`.
  // Fallback to 1.
  let best: OutlineNode | null = null;
  for (const n of outline) {
    if (n.byteRange.start <= byteRange.start && n.page !== undefined) best = n;
  }
  return best?.page ?? 1;
}
```

```tsx
// apps/web/src/components/citation/pdf-page-view.tsx
import { useEffect, useRef } from 'react';
import { getDocument, GlobalWorkerOptions } from 'pdfjs-dist';

GlobalWorkerOptions.workerSrc = new URL('pdfjs-dist/build/pdf.worker.min.mjs', import.meta.url).toString();

export function PdfPageView({
  url,
  pageNumber,
  highlightFraction,
}: {
  url: string;
  pageNumber: number;
  highlightFraction?: { top: number; height: number };
}) {
  const ref = useRef<HTMLCanvasElement | null>(null);

  useEffect(() => {
    let cancelled = false;
    (async () => {
      const doc = await getDocument(url).promise;
      const page = await doc.getPage(pageNumber);
      const canvas = ref.current;
      if (!canvas || cancelled) return;
      const viewport = page.getViewport({ scale: 1.5 });
      canvas.width = viewport.width;
      canvas.height = viewport.height;
      const ctx = canvas.getContext('2d');
      if (!ctx) return;
      await page.render({ canvasContext: ctx, viewport }).promise;
    })();
    return () => { cancelled = true; };
  }, [url, pageNumber]);

  return (
    <div className="relative">
      <canvas ref={ref} className="block w-full" />
      {highlightFraction ? (
        <div
          className="pointer-events-none absolute left-0 right-0 bg-accent/25 ring-2 ring-accent"
          style={{ top: `${highlightFraction.top * 100}%`, height: `${highlightFraction.height * 100}%` }}
        />
      ) : null}
    </div>
  );
}
```

```ts
import type { AnswerEvent } from '@package/contracts/chat';
import type { AnswerSegment } from '@package/contracts/shared';
import { useEventStream } from '../../lib/use-event-stream.ts';

export function useAnswerStream(turnId: string | null) {
  const { events, done, error } = useEventStream<AnswerEvent>(
    turnId ? `/rpc/turns/${turnId}/answer/events` : null,
    (raw) => raw as AnswerEvent,
  );
  const segments: AnswerSegment[] = [];
  for (const e of events) if (e.kind === 'AnswerSegment') segments[e.index] = e.segment;
  const finished = events.some((e) => e.kind === 'AnswerFinished');
  return { segments, finished: finished || done, error };
}
```

```tsx
// apps/web/src/routes/wiki.tsx
import { useMutation, useQuery } from '@tanstack/react-query';
import { useEffect, useState } from 'react';
import { Link, useParams } from 'react-router-dom';
import { Button } from '../components/ui/button.tsx';
import { CompileTheater } from '../components/compile-theater/compile-theater.tsx';
import { LoadingState } from '../components/states/loading.tsx';
import { orpc } from '../lib/orpc.ts';

export function WikiRoute() {
  const { wikiId = '' } = useParams(); // here `wikiId` actually means folderId on first visit
  const [compileRunId, setCompileRunId] = useState<string | null>(null);

  const start = useMutation({
    ...orpc.wiki.startCompile.mutationOptions(),
    onSuccess: (data) => setCompileRunId(data.compileRunId),
  });

  useEffect(() => { if (!compileRunId) start.mutate({ folderId: wikiId } as never); /* eslint-disable-line */ }, [wikiId]);

  const wiki = useQuery(orpc.wiki.getWiki.queryOptions({ input: { id: wikiId } as never, enabled: !!wikiId } as never));
  const pages = useQuery(orpc.wiki.listPages.queryOptions({ input: { wikiId, limit: 100 } as never, enabled: !!wikiId } as never));

  if (wiki.isPending) return <LoadingState rows={4} />;

  return (
    <main className="mx-auto max-w-7xl px-6 py-8">
      <header className="flex items-baseline justify-between">
        <div>
          <h1 className="font-serif text-3xl">{wiki.data?.folderId}</h1>
          <p className="text-sm text-muted-foreground">PageTypes: {wiki.data?.schema.pageTypes.map((p) => p.name).join(', ')}</p>
        </div>
        <div className="flex gap-2">
          <Button asChild variant="ghost"><Link to={`/chat/new?wikiId=${wikiId}`}>Chat</Link></Button>
          <Button asChild variant="outline"><Link to={`/wiki/${wikiId}/lint`}>Lint</Link></Button>
        </div>
      </header>

      {compileRunId ? <CompileTheater compileRunId={compileRunId} /> : null}

      <section className="mt-12 grid grid-cols-3 gap-6">
        {wiki.data?.schema.pageTypes.map((pt) => (
          <div key={pt.name}>
            <p className="font-mono text-xs uppercase tracking-widest text-accent">{pt.name}</p>
            <ul className="mt-2 space-y-1">
              {pages.data?.items.filter((p) => p.pageType === pt.name).map((p) => (
                <li key={p.id}>
                  <Link to={`/wiki/${wikiId}/page/${p.id}`} className="text-sm hover:underline">{p.title}</Link>
                </li>
              ))}
            </ul>
          </div>
        ))}
      </section>
    </main>
  );
}
```
