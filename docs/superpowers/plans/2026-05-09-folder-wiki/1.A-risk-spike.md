// scripts/spike/src/drive.ts
import { google } from 'googleapis';
import { mkdir, readFile, writeFile } from 'node:fs/promises';
import { createServer } from 'node:http';
import { resolve } from 'node:path';

const TOKEN_PATH = resolve(import.meta.dirname, '../.tokens.json');
const SCOPES = ['https://www.googleapis.com/auth/drive.readonly'];

const oauth = () =>
  new google.auth.OAuth2(
    process.env.GOOGLE_CLIENT_ID,
    process.env.GOOGLE_CLIENT_SECRET,
    process.env.GOOGLE_REDIRECT_URI,
  );

const auth = async () => {
  const o = oauth();
  const url = o.generateAuthUrl({ access_type: 'offline', scope: SCOPES });
  console.log('Open this URL:\n', url);
  const code = await new Promise<string>((res, rej) => {
    const srv = createServer((req, _) => {
      const u = new URL(req.url ?? '/', 'http://127.0.0.1:8765');
      if (u.pathname !== '/callback') return;
      const c = u.searchParams.get('code');
      if (!c) return rej(new Error('no code'));
      srv.close();
      res(c);
    }).listen(8765);
  });
  const { tokens } = await o.getToken(code);
  await writeFile(TOKEN_PATH, JSON.stringify(tokens, null, 2));
  console.log('Saved tokens to .tokens.json');
};

const list = async (folderId: string) => {
  const o = oauth();
  o.setCredentials(JSON.parse(await readFile(TOKEN_PATH, 'utf8')));
  const drive = google.drive({ version: 'v3', auth: o });
  const r = await drive.files.list({
    q: `'${folderId}' in parents and trashed = false`,
    fields: 'files(id,name,mimeType,modifiedTime,size)',
    pageSize: 50,
  });
  console.log(JSON.stringify(r.data.files, null, 2));
};

export const downloadPdf = async (fileId: string, outPath: string) => {
  const o = oauth();
  o.setCredentials(JSON.parse(await readFile(TOKEN_PATH, 'utf8')));
  const drive = google.drive({ version: 'v3', auth: o });
  await mkdir(resolve(import.meta.dirname, '../fixtures/drive'), { recursive: true });
  const r = await drive.files.get(
    { fileId, alt: 'media' },
    { responseType: 'arraybuffer' },
  );
  await writeFile(outPath, Buffer.from(r.data as ArrayBuffer));
};

const cmd = process.argv[2];
if (cmd === 'auth') void auth();
else if (cmd === 'list') void list(process.argv[3] ?? '');
```

## Task 3: PDF.js extraction with byte-range tracking

**Why:** The whole product hinges on `Span` byte-ranges aligning to actual PDF text after extraction. If extraction drifts, every Citation drifts.

**Files:**
- Create: `scripts/spike/src/extract.ts`

- [ ] **Step 1: Implement extraction**

```ts
// scripts/spike/src/extract.ts
import { getDocument } from 'pdfjs-dist/legacy/build/pdf.mjs';
import { createHash } from 'node:crypto';
import { readFile } from 'node:fs/promises';

export interface ExtractedSpan {
  start: number;
  end: number;
  page: number;
  text: string;
}

export interface Extraction {
  text: string;
  contentHash: string;
  spans: ExtractedSpan[];
  pageCount: number;
}

export const extractPdf = async (path: string): Promise<Extraction> => {
  const buf = await readFile(path);
  const contentHash = `sha256:${createHash('sha256').update(buf).digest('hex')}`;
  const doc = await getDocument({ data: new Uint8Array(buf) }).promise;
  const pageCount = doc.numPages;
  let cursor = 0;
  const spans: ExtractedSpan[] = [];
  let full = '';
  for (let p = 1; p <= pageCount; p++) {
    const page = await doc.getPage(p);
    const content = await page.getTextContent();
    for (const item of content.items) {
      if ('str' in item) {
        const text = item.str;
        if (!text) continue;
        spans.push({ start: cursor, end: cursor + text.length, page: p, text });
        cursor += text.length;
        full += text + ' ';
        cursor += 1; // for the trailing space
      }
    }
    full += '\n';
    cursor += 1;
  }
  return { text: full, contentHash, spans, pageCount };
};
```

// scripts/spike/src/measure.ts
import Anthropic from '@anthropic-ai/sdk';

export interface CallMeasurement {
  agent: string;
  model: string;
  ms: number;
  inputTokens: number;
  outputTokens: number;
  usd: number;
}

const PRICING_USD_PER_M_TOKENS: Record<string, { input: number; output: number }> = {
  'claude-haiku-4-5-20251001': { input: 0.8, output: 4 },
  'claude-sonnet-4-6': { input: 3, output: 15 },
  'claude-opus-4-7': { input: 15, output: 75 },
};

export const measure = async <T>(
  agent: string,
  model: string,
  fn: (client: Anthropic) => Promise<{ result: T; usage: { input_tokens: number; output_tokens: number } }>,
): Promise<{ result: T; m: CallMeasurement }> => {
  const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });
  const t0 = performance.now();
  const { result, usage } = await fn(client);
  const ms = performance.now() - t0;
  const p = PRICING_USD_PER_M_TOKENS[model] ?? { input: 0, output: 0 };
  const usd =
    (usage.input_tokens / 1_000_000) * p.input +
    (usage.output_tokens / 1_000_000) * p.output;
  return {
    result,
    m: { agent, model, ms, inputTokens: usage.input_tokens, outputTokens: usage.output_tokens, usd },
  };
};
```

- [ ] **Step 2: Implement agents.ts (SchemaInferrer + Drafter + Verifier only)**

```ts
// scripts/spike/src/agents.ts
import { z } from 'zod';
import { measure } from './measure.ts';

const Schema = z.object({
  pageTypes: z.array(z.object({ name: z.string(), description: z.string() })),
  relations: z.array(
    z.object({
      name: z.string(),
      from: z.string(),
      to: z.string(),
      cardinality: z.enum(['one-to-one', 'one-to-many', 'many-to-one', 'many-to-many']),
    }),
  ),
  reason: z.string(),
});

export const inferSchema = async (sourceTexts: string[]) =>
  measure('SchemaInferrer', 'claude-sonnet-4-6', async (c) => {
    const r = await c.messages.create({
      model: 'claude-sonnet-4-6',
      max_tokens: 1500,
      system:
        'You are SchemaInferrer. Read the first 5–10 sources from a folder and return a typed wiki schema. Return ONLY JSON conforming to: { pageTypes: { name (PascalCase), description }[], relations: { name, from, to, cardinality }[], reason: string }.',
      messages: [
        {
          role: 'user',
          content: sourceTexts
            .map((t, i) => `<source idx="${i}">\n${t.slice(0, 8000)}\n</source>`)
            .join('\n'),
        },
      ],
    });
    const json = (r.content[0] as { type: 'text'; text: string }).text;
    const parsed = Schema.parse(JSON.parse(extractJson(json)));
    return { result: parsed, usage: r.usage };
  });

const Draft = z.object({
  title: z.string(),
  body: z.string(),
  claims: z.array(
    z.object({
      paragraphId: z.string(),
      claimText: z.string(),
      citationLabels: z.array(z.string()).min(1),
    }),
  ),
});

export const draftPage = async (sourceText: string, sourceFilename: string) =>
  measure('Drafter', 'claude-sonnet-4-6', async (c) => {
    const r = await c.messages.create({
      model: 'claude-sonnet-4-6',
      max_tokens: 2500,
      system:
        'You are Drafter. Compose a SummaryPage in markdown for the given source. Return ONLY JSON: { title, body (markdown), claims: { paragraphId, claimText, citationLabels[] }[] }. Each claim must reference at least one citation label.',
      messages: [
        {
          role: 'user',
          content: `Source filename: ${sourceFilename}\n\n${sourceText.slice(0, 12000)}`,
        },
      ],
    });
    const txt = (r.content[0] as { type: 'text'; text: string }).text;
    return { result: Draft.parse(JSON.parse(extractJson(txt))), usage: r.usage };
  });

const Verdict = z.object({
  verdict: z.enum(['supported', 'unsupported', 'contradicted']),
  evidenceText: z.string(),
  correction: z.object({ replacementText: z.string() }).optional(),
});

export const verifyClaim = async (claim: string, citedText: string) =>
  measure('Verifier', 'claude-opus-4-7', async (c) => {
    const r = await c.messages.create({
      model: 'claude-opus-4-7',
      max_tokens: 800,
      temperature: 0,
      system:
        'You are Verifier. Decide if the cited text supports, fails to support, or contradicts the claim. Return ONLY JSON: { verdict: "supported"|"unsupported"|"contradicted", evidenceText, correction?: { replacementText } }.',
      messages: [
        {
          role: 'user',
          content: `Claim:\n${claim}\n\nCited text:\n${citedText}`,
        },
      ],
    });
    const txt = (r.content[0] as { type: 'text'; text: string }).text;
    return { result: Verdict.parse(JSON.parse(extractJson(txt))), usage: r.usage };
  });

const extractJson = (s: string): string => {
  const m = s.match(/```json\s*([\s\S]+?)\s*```/);
  if (m?.[1]) return m[1];
  const i = s.indexOf('{');
  const j = s.lastIndexOf('}');
  return s.slice(i, j + 1);
};
```

<!-- scripts/spike/fixtures/contradiction-claim.md -->
Q3 NRR was **110%**, exceeding the 106% target.

- [ ] **Step 2: Implement run.ts**

```ts
// scripts/spike/src/run.ts
import { mkdir, writeFile } from 'node:fs/promises';
import { resolve } from 'node:path';
import { downloadPdf } from './drive.ts';
import { extractPdf } from './extract.ts';
import { draftPage, inferSchema, verifyClaim } from './agents.ts';
import type { CallMeasurement } from './measure.ts';

const FOLDER_ID = process.env.SPIKE_FOLDER_ID;
const FILES: Array<{ id: string; name: string }> = JSON.parse(
  process.env.SPIKE_FILES ?? '[]',
);

const OUT = resolve(import.meta.dirname, '../out');

const fmt = (m: CallMeasurement) =>
  `${m.agent.padEnd(15)} ${m.model.padEnd(28)} ${m.ms.toFixed(0).padStart(6)} ms ${m.inputTokens.toString().padStart(7)} in ${m.outputTokens.toString().padStart(6)} out  $${m.usd.toFixed(4)}`;

const main = async () => {
  if (!FOLDER_ID || FILES.length === 0)
    throw new Error('Set SPIKE_FOLDER_ID and SPIKE_FILES env vars');

  await mkdir(OUT, { recursive: true });
  const measurements: CallMeasurement[] = [];

  // 1. Download + extract 5 PDFs
  const sources = await Promise.all(
    FILES.slice(0, 5).map(async (f) => {
      const path = resolve(import.meta.dirname, `../fixtures/drive/${f.name}`);
      await downloadPdf(f.id, path);
      const e = await extractPdf(path);
      return { ...f, ...e };
    }),
  );
  console.log('Extracted', sources.length, 'sources, total chars',
    sources.reduce((a, s) => a + s.text.length, 0));

  // 2. Schema inference
  const { result: schema, m: schemaM } = await inferSchema(sources.map((s) => s.text));
  measurements.push(schemaM);
  console.log('\nSchema:\n', JSON.stringify(schema, null, 2));

  // 3. Draft a SummaryPage per source
  const drafts = [];
  for (const s of sources) {
    const { result, m } = await draftPage(s.text, s.name);
    measurements.push(m);
    drafts.push({ source: s.name, ...result });
  }

  // 4. Verify each claim against the cited text. For the spike,
  //    cite-text is approximated by searching `s.text` for citationLabels;
  //    in production the Citation has byte_range. We're testing
  //    Verifier behavior, not byte-range mechanics.
  const findings: { claim: string; verdict: string; evidenceText: string }[] = [];
  for (const draft of drafts) {
    const source = sources.find((s) => s.name === draft.source);
    if (!source) continue;
    for (const claim of draft.claims.slice(0, 3)) {
      const cited = claim.citationLabels
        .map((l) => spanForLabel(source.text, l))
        .join('\n\n');
      const { result, m } = await verifyClaim(claim.claimText, cited);
      measurements.push(m);
      findings.push({ claim: claim.claimText, verdict: result.verdict, evidenceText: result.evidenceText });
    }
  }

  // 5. Determinism — run the planted contradiction 10×
  const planted = await runPlantedContradictionTimes(10);
  measurements.push(...planted.measurements);

  // 6. Report
  const totalUsd = measurements.reduce((a, m) => a + m.usd, 0);
  const totalMs = measurements.reduce((a, m) => a + m.ms, 0);
  const lines = [
    '# folder-wiki risk spike report',
    '',
    `Run at ${new Date().toISOString()}`,
    `Sources: ${sources.length} PDFs, ${sources.reduce((a, s) => a + s.text.length, 0)} chars total`,
    `Schema PageTypes: ${schema.pageTypes.map((p) => p.name).join(', ')}`,
    `Schema Relations: ${schema.relations.map((r) => `${r.from}-${r.name}->${r.to}`).join(', ')}`,
    '',
    '## per-call measurements',
    '```',
    'agent           model                          ms    in_tok out_tok  usd',
    ...measurements.map(fmt),
    '```',
    '',
    `**Total wallclock:** ${(totalMs / 1000).toFixed(1)}s`,
    `**Total cost:** $${totalUsd.toFixed(4)}`,
    '',
    `## findings (${findings.length})`,
    ...findings.map((f) => `- [${f.verdict}] ${f.claim} — ${f.evidenceText.slice(0, 120)}…`),
    '',
    '## planted-contradiction determinism',
    `Verifier ran ${planted.runs} times on the planted "NRR=110% (cited span says 105%)" pair.`,
    `Verdicts: ${planted.verdicts.join(', ')}`,
    `Caught (verdict ≠ "supported") in ${planted.caught} / ${planted.runs} runs.`,
  ].join('\n');

  await writeFile(resolve(OUT, 'report.md'), lines);
  console.log('\nWrote', resolve(OUT, 'report.md'));
};

const spanForLabel = (text: string, label: string): string => {
  const i = text.toLowerCase().indexOf(label.split(',')[0]?.toLowerCase() ?? label.toLowerCase());
  if (i < 0) return text.slice(0, 600);
  return text.slice(Math.max(0, i - 200), i + 800);
};

const runPlantedContradictionTimes = async (n: number) => {
  const claim = 'Q3 NRR was 110%, exceeding the 106% target.';
  const citedText = 'Q3 NRR landed at 105%, four points below the 109% target. The board flagged this as a watch-item.';
  const verdicts: string[] = [];
  const measurements = [];
  for (let i = 0; i < n; i++) {
    const { result, m } = await verifyClaim(claim, citedText);
    verdicts.push(result.verdict);
    measurements.push(m);
  }
  return {
    runs: n,
    verdicts,
    caught: verdicts.filter((v) => v !== 'supported').length,
    measurements,
  };
};

void main();
```
