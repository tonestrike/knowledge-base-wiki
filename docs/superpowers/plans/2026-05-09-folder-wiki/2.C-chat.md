# Slice 2.C — chat (Generative artifact answers)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make chat real. A user asks a question; Researcher (Haiku 4.5) reads the wiki to gather typed findings; Synthesizer (Sonnet 4.6) composes an `Answer` as a stream of `AnswerSegment`s — interleaving prose, citation chips, and one or more `Artifact` components from the closed 8-component registry. Every `Citation.span` is hash-verified against the actual source bytes — no fabrication. A `ConversationDO` orchestrates streaming with SSE; persistent state lives in D1.

**Architecture:** DDD parallel to 2.B. `domain/` carries `Conversation`, `Turn`, `Answer`, `AnswerSegment`. `application/` has `open-conversation`, `ask`, `research-question`, `synthesize-answer`, `verify-citation`, `get-turn`, `list-turns`, `get-conversation`, `list-conversations`. `infrastructure/` adds Anthropic structured-output for the Synthesizer (the single most strict prompt in the system — its JSON schema enforces the closed `ArtifactKind` registry), the `ConversationDO`, D1 repos, and a `WikiReader` adapter that reads pages via `@package/contracts/wiki` over the network so `domains/chat` never imports `domains/wiki`.

**Tech Stack:** `@anthropic-ai/sdk` 0.40.x, oRPC client (used inside the chat domain via the typed contract — read-only) for cross-context wiki reads, D1, Cloudflare Durable Objects, R2 (read-only access to ingestion source text for hash-check). Cross-domain coordination is via `@package/contracts` only.

```ts
// packages/domains/chat/src/domain/turn.ts
import type { AnswerSegment, ConversationId, TurnId } from '@package/contracts/shared';

export interface Turn {
  readonly id: TurnId;
  readonly conversationId: ConversationId;
  readonly question: string;
  readonly answer: ReadonlyArray<AnswerSegment>;
  readonly createdAt: string;
  readonly finishedAt?: string;
}

export const Turn = {
  start(props: { id: TurnId; conversationId: ConversationId; question: string; createdAt: string }): Turn {
    if (!props.question.trim()) throw new Error('question is empty');
    return Object.freeze({ ...props, answer: Object.freeze([]) });
  },
  appendSegment(t: Turn, seg: AnswerSegment): Turn {
    if (t.finishedAt) throw new Error('cannot append to a finished Turn');
    return Object.freeze({ ...t, answer: Object.freeze([...t.answer, seg]) });
  },
  finish(t: Turn, at: string): Turn {
    if (t.finishedAt) return t;
    return Object.freeze({ ...t, finishedAt: at });
  },
};
```

```ts
import { turnId, type ConversationId, type TurnId } from '@package/contracts/shared';
import { Turn } from '../domain/turn.ts';
import type { ChatDeps } from './ports.ts';

export async function ask(
  deps: ChatDeps,
  input: { conversationId: ConversationId; question: string },
): Promise<{ turnId: TurnId }> {
  const c = await deps.conversations.findById(input.conversationId);
  if (!c) throw new Error(`Conversation not found: ${input.conversationId}`);
  const id = turnId(deps.newId());
  const turn = Turn.start({ id, conversationId: input.conversationId, question: input.question, createdAt: deps.now().toISOString() });
  await deps.turns.insert(turn);
  await deps.dispatcher.start({ conversationId: c.id, turnId: id, wikiId: c.wikiId, question: input.question });
  return { turnId: id };
}
```

```ts
// packages/domains/chat/src/application/research-question.ts
import { z } from 'zod';
import type { AnthropicClient, WikiPageSummary, WikiReader } from './ports.ts';
import type { Citation, WikiId, WikiPageId } from '@package/contracts/shared';

export const RESEARCHER_MODEL = 'claude-haiku-4-5-20251001';

const ResearchOutput = z.object({
  findings: z.array(
    z.object({
      wikiPageId: z.string().uuid(),
      quoteText: z.string().min(1).max(800),
      citationIds: z.array(z.string().uuid()).min(1),
    }),
  ).max(20),
});

const SYSTEM = `You are Researcher. Read the candidate wiki pages below. For each that contains information relevant to the question, emit one or more findings. Each finding MUST quote the relevant text verbatim from the page body, AND name the citation IDs (provided per page) that support it. Do not invent citation IDs. Return ONLY JSON: { findings: [{ wikiPageId, quoteText, citationIds[] }] }.`;

export async function researchQuestion(
  deps: { anthropic: AnthropicClient; wikiReader: WikiReader },
  input: { wikiId: WikiId; question: string },
): Promise<{ pages: WikiPageSummary[]; findings: Array<{ wikiPageId: WikiPageId; quoteText: string; citationIds: string[]; citations: Citation[] }> }> {
  const candidates = await deps.wikiReader.searchPages({ wikiId: input.wikiId, query: input.question, limit: 8 });

  const block = candidates.map((p) => {
    const citationsBlock = p.citations.map((c) => `<citation id="${c.id}" label="${c.label}"/>`).join('\n');
    return `<page id="${p.id}" type="${p.pageType ?? ''}" title="${p.title}">\n${citationsBlock}\n${p.body}\n</page>`;
  }).join('\n');

  const { result } = await deps.anthropic.jsonComplete({
    model: RESEARCHER_MODEL,
    system: SYSTEM,
    messages: [{ role: 'user', content: `Question: ${input.question}\n\nPages:\n${block}` }],
    schema: ResearchOutput,
    maxTokens: 1500,
    temperature: 0.1,
  });

  const byPageId = new Map(candidates.map((p) => [p.id, p]));
  const findings = result.findings.flatMap((f) => {
    const page = byPageId.get(f.wikiPageId as WikiPageId);
    if (!page) return [];
    const cites = f.citationIds
      .map((id) => page.citations.find((c) => c.id === id))
      .filter((c): c is Citation => Boolean(c));
    if (cites.length === 0) return [];
    return [{ wikiPageId: page.id, quoteText: f.quoteText, citationIds: f.citationIds, citations: cites }];
  });
  return { pages: candidates, findings };
}
```

```ts
import type { AnswerEvent } from '@package/contracts/chat';
import { ArtifactKind, type AnswerSegment, type Citation, type TurnId } from '@package/contracts/shared';
import type { AnthropicClient, SourceHashVerifier } from './ports.ts';

export const SYNTH_MODEL = 'claude-sonnet-4-6';

// The Synthesizer's structured-output schema. Closed registry — every Artifact
// kind is enumerated. `citationIdsByProp` tells the renderer which prop value
// in props uses which citation; the registry's typed prop schema is enforced
// at parse time.
const ArtifactKindEnum = ArtifactKind;
const ArtifactProps = z.union([
  z.object({ kind: z.literal('ComparisonTable'), props: z.object({
    columns: z.array(z.string()).min(2).max(8),
    rows: z.array(z.object({ cells: z.array(z.object({ value: z.string(), citationId: z.string().uuid().optional() })).min(2).max(8) })).max(40),

  z.object({ kind: z.literal('Timeline'), props: z.object({
    events: z.array(z.object({ at: z.string(), label: z.string(), description: z.string().optional(), citationId: z.string().uuid().optional() })).max(40),
  }) }),
  z.object({ kind: z.literal('LineChart'), props: z.object({
    xLabel: z.string(), yLabel: z.string(),
    series: z.array(z.object({ name: z.string(), points: z.array(z.object({ x: z.union([z.string(), z.number()]), y: z.number(), citationId: z.string().uuid().optional() })).max(60) })).max(6),
  }) }),
  z.object({ kind: z.literal('BarChart'), props: z.object({
    xLabel: z.string(), yLabel: z.string(),
    bars: z.array(z.object({ x: z.union([z.string(), z.number()]), y: z.number(), citationId: z.string().uuid().optional() })).max(40),
  }) }),
  z.object({ kind: z.literal('KeyMetric'), props: z.object({
    label: z.string(), value: z.string(), delta: z.string().optional(), trend: z.enum(['up', 'down', 'flat']).optional(), citationId: z.string().uuid().optional(),
  }) }),
  z.object({ kind: z.literal('CodeBlock'), props: z.object({ language: z.string(), source: z.string().max(20_000) }) }),
  z.object({ kind: z.literal('Quote'), props: z.object({ text: z.string().max(2000), attribution: z.string().optional(), citationId: z.string().uuid().optional() }) }),
  z.object({ kind: z.literal('Markdown'), props: z.object({ body: z.string().max(20_000) }) }),
]);

const SynthOutput = z.object({
  segments: z.array(
    z.discriminatedUnion('kind', [
      z.object({ kind: z.literal('prose'), text: z.string().min(1).max(4000) }),
      z.object({ kind: z.literal('citation'), citationId: z.string().uuid() }),
      z.object({ kind: z.literal('artifact'), artifact: ArtifactProps.and(z.object({ citationIds: z.array(z.string().uuid()).min(1) })) }),
    ]),
  ).min(1).max(40),
});

const SYSTEM = `You are Synthesizer. Compose an Answer to the user's question using the provided findings.

Hard rules:
- Every prose claim that asserts a fact MUST be paired with a "citation" segment immediately after it.
- Where a comparison, trend, code, quote, or single metric is the clearest form, emit ONE "artifact" segment using the closed registry: ${ArtifactKindEnum.options.join(' | ')}.
- Each artifact MUST list the citationIds it uses.
- Only use citationIds that appear in the provided findings. Do NOT invent.
- The first segment is usually short prose introducing the answer; the last segment is a takeaway sentence.

Return ONLY JSON: { segments: AnswerSegment[] } where AnswerSegment is { kind: 'prose', text } | { kind: 'citation', citationId } | { kind: 'artifact', artifact: { kind, props, citationIds[] } }.`;

export interface SynthesizeAnswerDeps {
  anthropic: AnthropicClient;
  sourceHashes: SourceHashVerifier;
  emit(event: AnswerEvent): Promise<void>;
}

export async function synthesizeAnswer(
  deps: SynthesizeAnswerDeps,
  input: {
    turnId: TurnId;
    question: string;
    findings: Array<{ quoteText: string; citationIds: string[]; citations: Citation[] }>;
  },
): Promise<{ segments: AnswerSegment[] }> {
  await deps.emit({ kind: 'AnswerStarted', turnId: input.turnId });

  const knownCitations = new Map<string, Citation>();
  for (const f of input.findings) for (const c of f.citations) knownCitations.set(c.id, c);

  const findingsBlock = input.findings.map((f) =>
    `<finding citationIds="${f.citationIds.join(',')}">\n${f.quoteText}\n</finding>`,
  ).join('\n');

  const { result } = await deps.anthropic.jsonComplete({
    model: SYNTH_MODEL,
    system: SYSTEM,
    messages: [{ role: 'user', content: `Question: ${input.question}\n\nFindings:\n${findingsBlock}` }],
    schema: SynthOutput,
    maxTokens: 3500,
    temperature: 0.4,
  });

  const out: AnswerSegment[] = [];
  let index = 0;

  for (const seg of result.segments) {
    let answerSeg: AnswerSegment;
    if (seg.kind === 'prose') {
      answerSeg = seg;
    } else if (seg.kind === 'citation') {
      const c = knownCitations.get(seg.citationId);
      if (!c) throw new Error(`Synthesizer cited unknown citation ${seg.citationId}`);
      const v = await deps.sourceHashes.verify(c);
      if (!v.ok) throw new Error(`Citation ${c.id} failed hash check: ${v.reason}`);
      answerSeg = { kind: 'citation', citation: c };
    } else {
      // artifact
      const cites = seg.artifact.citationIds.map((id) => {
        const c = knownCitations.get(id);
        if (!c) throw new Error(`Synthesizer cited unknown citation ${id}`);
        return c;
      });
      for (const c of cites) {
        const v = await deps.sourceHashes.verify(c);
        if (!v.ok) throw new Error(`Citation ${c.id} failed hash check: ${v.reason}`);
      }
      answerSeg = { kind: 'artifact', artifact: { ...seg.artifact, citations: cites } as never };
    }
    out.push(answerSeg);
    await deps.emit({ kind: 'AnswerSegment', turnId: input.turnId, index, segment: answerSeg });
    index++;
  }

  await deps.emit({ kind: 'AnswerFinished', turnId: input.turnId });
  return { segments: out };
}

  });
```

```ts
// packages/domains/chat/src/application/verify-citation.ts
import type { Citation, SourceId } from '@package/contracts/shared';
import type { SourceHashVerifier } from './ports.ts';

export interface SourceHashVerifierDeps {
  readSourceText(sourceId: SourceId): Promise<string | null>;
  sha256Hex(slice: string): Promise<string>;
}

export const createMemorySourceHashVerifier = (deps: SourceHashVerifierDeps): SourceHashVerifier => ({
  async verify(citation: Citation) {
    const text = await deps.readSourceText(citation.span.sourceId);
    if (text == null) return { ok: false, reason: `source text not found: ${citation.span.sourceId}` };
    if (citation.span.byteRange.end > text.length) {
      return { ok: false, reason: `byte range overruns source text (${citation.span.byteRange.end} > ${text.length})` };
    }
    const slice = text.slice(citation.span.byteRange.start, citation.span.byteRange.end);
    const hash = await deps.sha256Hex(slice);
    if (hash !== citation.span.contentHash) {
      return { ok: false, reason: `hash mismatch: ${hash} vs ${citation.span.contentHash}` };
    }
    return { ok: true };
  },
});
```

> Note: in production this verifies the *whole-source* content hash, not the slice. We chose slice-hash as a stricter invariant — it catches the case where a Citation's byte range is correct but the underlying source has shifted. The contracts layer uses content_hash on the whole `Source`; chat verifies the slice produces the recorded hash. Document this divergence in the slice's PR description.

```ts
import { createORPCClient } from '@orpc/client';
import { RPCLink } from '@orpc/client/fetch';
import type { ContractRouterClient } from '@orpc/contract';
import type { Contract } from '@package/contracts';
import type { WikiPageId, WikiId } from '@package/contracts/shared';
import type { WikiPageSummary, WikiReader } from '../application/ports.ts';

export const createOrpcWikiReader = (rpcUrl: string): WikiReader => {
  const link = new RPCLink({ url: rpcUrl });
  const client: ContractRouterClient<Contract> = createORPCClient(link);

  return {
    async searchPages({ wikiId, query, limit }) {
      // v1: list all pages and rank locally by simple substring match.
      // Phase 3.x can swap in a real `searchPages` procedure; the chat side
      // doesn't need to know.
      const out: WikiPageSummary[] = [];
      let cursor: string | undefined;
      while (out.length < 200) {
        const page = await client.wiki.listPages({ wikiId, limit: 100, cursor });
        for (const p of page.items) {
          if (out.length >= 200) break;
          out.push({
            id: p.id, wikiId: p.wikiId, title: p.title, pageType: p.pageType,
            body: p.body, citations: p.citations,
          });
        }
        if (!page.nextCursor) break;
        cursor = page.nextCursor;
      }
      const q = query.toLowerCase();
      return out
        .map((p) => ({ p, score: scoreMatch(p, q) }))
        .filter((x) => x.score > 0)
        .sort((a, b) => b.score - a.score)
        .slice(0, limit)
        .map((x) => x.p);
    },
    async getPage(id: WikiPageId) {
      const p = await client.wiki.getPage({ id });
      return { id: p.id, wikiId: p.wikiId, title: p.title, pageType: p.pageType, body: p.body, citations: p.citations };
    },
  };
};

const scoreMatch = (p: WikiPageSummary, q: string): number => {
  const tokens = q.split(/\s+/).filter(Boolean);
  let s = 0;
  for (const t of tokens) {
    if (p.title.toLowerCase().includes(t)) s += 5;
    if (p.body.toLowerCase().includes(t)) s += 1;
  }
  return s;
};
```

```ts
import type { AnswerEvent } from '@package/contracts/chat';
import { researchQuestion } from '../../application/research-question.ts';
import { synthesizeAnswer } from '../../application/synthesize-answer.ts';
import type { ChatRuntimeDeps } from '../../application/ports.ts';

interface State { storage: { get<T>(k: string): Promise<T | undefined>; put(k: string, v: unknown): Promise<void> }; waitUntil(p: Promise<unknown>): void }
interface Env { [k: string]: unknown }

interface AskCommand {
  kind: 'ask';
  conversationId: string;
  turnId: string;
  wikiId: string;
  question: string;
}

const tapeKey = (turnId: string) => `tape:${turnId}`;

export interface ConversationDOFactoryArgs {
  buildDeps: (env: Env, emit: (e: AnswerEvent) => Promise<void>) => ChatRuntimeDeps;
}

export const createConversationDOClass = ({ buildDeps }: ConversationDOFactoryArgs) =>
  class ConversationDO {
    private subs = new Map<string, Set<(e: AnswerEvent) => void>>();

    constructor(private state: State, private env: Env) {}

    async fetch(req: Request): Promise<Response> {
      const url = new URL(req.url);
      if (url.pathname === '/ask' && req.method === 'POST') {
        const cmd = (await req.json()) as AskCommand;
        this.state.waitUntil(this.run(cmd));
        return new Response('ok', { status: 202 });
      }
      if (url.pathname.startsWith('/subscribe/') && req.method === 'GET') {
        const turnId = url.pathname.split('/').pop()!;
        const stream = new ReadableStream({
          start: (controller) => {
            const send = (e: AnswerEvent) => controller.enqueue(`data: ${JSON.stringify(e)}\n\n`);
            const set = this.subs.get(turnId) ?? new Set();
            set.add(send);
            this.subs.set(turnId, set);
            this.state.storage.get<AnswerEvent[]>(tapeKey(turnId)).then((tape) => {
              for (const e of tape ?? []) send(e);
            });
          },
        });
        return new Response(stream, { headers: { 'content-type': 'text/event-stream' } });
      }
      return new Response('not found', { status: 404 });
    }

    private async run(cmd: AskCommand) {
      const tape: AnswerEvent[] = [];
      const emit = async (e: AnswerEvent) => {
        tape.push(e);
        await this.state.storage.put(tapeKey(cmd.turnId), tape);
        for (const send of this.subs.get(cmd.turnId) ?? []) send(e);
      };

      const deps = buildDeps(this.env, emit);
      try {
        const { findings } = await researchQuestion(deps, { wikiId: cmd.wikiId as never, question: cmd.question });
        await synthesizeAnswer({ anthropic: deps.anthropic, sourceHashes: deps.sourceHashes, emit }, {
          turnId: cmd.turnId as never,
          question: cmd.question,
          findings,
        });
        // Domain event for the wiki context — see spec §2.3
        await deps.eventBus.publish({
          name: 'AnswerProduced',
          occurredAt: deps.now().toISOString(),
          payload: { conversationId: cmd.conversationId as never, turnId: cmd.turnId as never, wikiId: cmd.wikiId as never },
        });
      } catch (err) {
        await emit({ kind: 'AnswerFailed', turnId: cmd.turnId as never, message: String(err) });
      }
    }
  };
```

```ts
  },
  async *subscribe(turnId) {
    // The conversationId is needed to find the DO; we look it up via the
    // turn repo at the DO call site. Here we expect the caller to have
    // bound the DO id at creation time and to pass it in via the URL.
    // For simplicity, the DO uses idFromName(conversationId) and clients
    // subscribe through the same name; chat handler resolves conversationId
    // from the Turn before calling this function.
    throw new Error('use cf-conversation-dispatcher.subscribeFor(conversationId, turnId)');
  },
});

export const subscribeFor = (
  ns: DurableObjectNamespace,
  args: { conversationId: string; turnId: string },
): AsyncIterable<AnswerEvent> => ({
  async *[Symbol.asyncIterator]() {
    const stub = ns.get(ns.idFromName(args.conversationId));
    const res = await stub.fetch(new Request(`https://do/subscribe/${args.turnId}`));
    if (!res.body) return;
    const reader = res.body.getReader();
    const decoder = new TextDecoder();
    let buf = '';
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buf += decoder.decode(value, { stream: true });
      let idx;
      while ((idx = buf.indexOf('\n\n')) !== -1) {
        const evt = buf.slice(0, idx).replace(/^data:\s*/m, '').trim();
        buf = buf.slice(idx + 2);
        if (evt) yield JSON.parse(evt) as AnswerEvent;
      }
    }
  },
});
```
