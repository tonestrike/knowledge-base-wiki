# Slice 2.D — verification (span-verifying lint loop)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`) syntax.

**Goal:** The reliability leap. A `LintRun` audits every `Claim` in a Wiki against the actual source bytes its citations point to. The `Verifier` agent runs on **Opus 4.7** (different family from the Sonnet-driven Compiler — defeats self-grading bias). Per Claim it emits a `Verdict` (`supported | unsupported | contradicted`) plus `evidenceText` and (for non-supported) a `Correction`. On `CompileFinished` from 2.B, an automatic full pass runs. A daily Cron Trigger samples 10% of existing wikis. The lint catches the planted contradiction in the demo folder on ≥10/10 runs.

**Architecture:** DDD parallel to 2.B/2.C. `domain/` carries `LintRun`, `LintFinding`, `Verdict`, `Correction`, `Audit`. `application/` has `lint-wiki`, `audit-claim`, `apply-correction`, plus reads. `infrastructure/` adds the Verifier prompt (strict structured output), the `LintRunDO` (concurrent per-Claim audit, cap 6), the Cron handler, and a `WikiClaimReader` that reads claims via `@package/contracts/wiki`. Subscribes to `CompileFinished` for auto-trigger; `CorrectionAccepted` events are published when `applyCorrection` succeeds (the wiki context will subscribe later to apply the patch — for v1, the chat layer just shows the corrected page).

**Tech Stack:** Anthropic SDK (Opus 4.7), Cloudflare Durable Objects, D1, Cron Triggers. Verifier `temperature: 0` and a fixed model id, locked.

// packages/domains/verification/src/domain/lint-run.ts
import type { LintRunId, WikiId } from '@package/contracts/shared';

export type LintRunStatus = 'pending' | 'running' | 'finished' | 'failed';

export interface LintRun {
  readonly id: LintRunId;
  readonly wikiId: WikiId;
  readonly status: LintRunStatus;
  readonly totalClaims: number;
  readonly audited: number;
  readonly unsupportedCount: number;
  readonly contradictedCount: number;
  readonly startedAt: string;
  readonly endedAt?: string;
  readonly failureMessage?: string;
}

export const LintRun = {
  start(props: { id: LintRunId; wikiId: WikiId; totalClaims: number; startedAt: string }): LintRun {
    return Object.freeze({
      ...props,
      status: 'pending' as const,
      audited: 0, unsupportedCount: 0, contradictedCount: 0,
    });
  },
  run(r: LintRun): LintRun {
    if (r.status !== 'pending') throw new Error(`cannot run from ${r.status}`);
    return Object.freeze({ ...r, status: 'running' });
  },
  tally(r: LintRun, delta: { auditedDelta: number; unsupportedDelta: number; contradictedDelta: number }): LintRun {
    return Object.freeze({
      ...r,
      audited: r.audited + delta.auditedDelta,
      unsupportedCount: r.unsupportedCount + delta.unsupportedDelta,
      contradictedCount: r.contradictedCount + delta.contradictedDelta,
    });
  },
  finish(r: LintRun, at: string): LintRun {
    if (r.status !== 'running') throw new Error(`cannot finish from ${r.status}`);
    return Object.freeze({ ...r, status: 'finished', endedAt: at });
  },
  fail(r: LintRun, message: string, at: string): LintRun {
    return Object.freeze({ ...r, status: 'failed', failureMessage: message, endedAt: at });
  },
};
```

import type { Citation, Claim, ClaimId, LintFindingId, LintRunId, WikiPageId } from '@package/contracts/shared';
import type { Correction } from './correction.ts';

export type Verdict = 'supported' | 'unsupported' | 'contradicted';

export interface LintFinding {
  readonly id: LintFindingId;
  readonly lintRunId: LintRunId;
  readonly wikiPageId: WikiPageId;
  readonly claimId: ClaimId;
  readonly claim: Claim;
  readonly verdict: Verdict;
  readonly evidenceText: string;
  readonly citedSpans: ReadonlyArray<Citation>;
  readonly correction?: Correction;
  readonly resolvedAt?: string;
}

export const LintFinding = {
  create(props: Omit<LintFinding, 'resolvedAt'>): LintFinding {
    if ((props.verdict === 'unsupported' || props.verdict === 'contradicted') && !props.correction) {
      throw new Error('non-supported verdicts must carry a Correction');
    }
    return Object.freeze({ ...props, citedSpans: Object.freeze([...props.citedSpans]) });
  },
  resolve(f: LintFinding, at: string): LintFinding {
    return Object.freeze({ ...f, resolvedAt: at });
  },
};

// packages/domains/verification/src/application/audit-claim.ts
import type { Claim } from '@package/contracts/shared';
import { Correction } from '../domain/correction.ts';
import type { AnthropicVerifier, SourceTextReader } from './ports.ts';

export async function auditClaim(
  deps: { verifier: AnthropicVerifier; sourceText: SourceTextReader },
  input: { claim: Claim },
): Promise<{
  verdict: 'supported' | 'unsupported' | 'contradicted';
  evidenceText: string;
  correction?: Correction;
}> {
  const slices: Array<{ citationId: string; sliceText: string }> = [];
  for (const c of input.claim.citations) {
    const slice = await deps.sourceText.readSlice({ sourceId: c.span.sourceId, byteRange: c.span.byteRange });
    slices.push({ citationId: c.id, sliceText: slice ?? '<missing>' });
  }
  const out = await deps.verifier.audit({ claim: input.claim, citedSlices: slices });
  if (out.verdict !== 'supported' && !out.correction) {
    throw new Error(`Verifier returned non-supported verdict without a Correction (claim ${input.claim.id})`);
  }
  return {
    verdict: out.verdict,
    evidenceText: out.evidenceText,
    correction: out.correction ? Correction.create(out.correction) : undefined,
  };
}
```

// packages/domains/verification/src/application/lint-wiki.ts
import type { LintRunId, WikiId } from '@package/contracts/shared';
import { lintFindingId } from '@package/contracts/shared';
import { LintFinding } from '../domain/lint-finding.ts';
import { LintRun } from '../domain/lint-run.ts';
import { auditClaim } from './audit-claim.ts';
import type { LintRuntimeDeps } from './ports.ts';

export async function lintWiki(
  deps: LintRuntimeDeps,
  input: { lintRunId: LintRunId; wikiId: WikiId },
): Promise<{ finishedAt: string }> {
  const startedAt = deps.now().toISOString();
  const all = await deps.claims.listClaimsForWiki(input.wikiId);
  let run = LintRun.start({ id: input.lintRunId, wikiId: input.wikiId, totalClaims: all.length, startedAt });
  await deps.runs.insert(run);
  await deps.emit({ kind: 'LintRunStarted', lintRunId: run.id, totalClaims: all.length });
  run = LintRun.run(run);
  await deps.runs.update(run);

  // Concurrency-capped fan-out
  const sem = createSemaphore(deps.concurrency);
  const findings = await Promise.all(all.map(async ({ wikiPageId, claim }) => {
    return sem(async () => {
      const out = await auditClaim({ verifier: deps.verifier, sourceText: deps.sourceText }, { claim });
      const finding = LintFinding.create({
        id: lintFindingId(deps.newId()),
        lintRunId: input.lintRunId,
        wikiPageId,
        claimId: claim.id,
        claim,
        verdict: out.verdict,
        evidenceText: out.evidenceText,
        citedSpans: claim.citations,
        correction: out.correction,
      });
      await deps.findings.insertMany([finding]);
      run = LintRun.tally(run, {
        auditedDelta: 1,
        unsupportedDelta: out.verdict === 'unsupported' ? 1 : 0,
        contradictedDelta: out.verdict === 'contradicted' ? 1 : 0,
      });
      await deps.runs.update(run);
      await deps.emit({
        kind: 'ClaimAudited',
        lintRunId: run.id,
        lintFindingId: finding.id,
        claimId: claim.id,
        wikiPageId,
        verdict: out.verdict,
      });
      return finding;
    });
  }));

  const finishedAt = deps.now().toISOString();
  run = LintRun.finish(run, finishedAt);
  await deps.runs.update(run);
  await deps.emit({
    kind: 'LintRunFinished',
    lintRunId: run.id,
    unsupportedCount: run.unsupportedCount,
    contradictedCount: run.contradictedCount,
  });

  return { finishedAt };
}

const createSemaphore = (max: number) => {
  let active = 0;
  const queue: Array<() => void> = [];
  const acquire = (): Promise<void> => new Promise((resolve) => {
    const tryGo = () => {
      if (active < max) { active++; resolve(); }
      else queue.push(tryGo);
    };
    tryGo();
  });
  return async <T>(fn: () => Promise<T>): Promise<T> => {
    await acquire();
    try {
      return await fn();
    } finally {
      active--;
      const next = queue.shift();
      next?.();
    }
  };
};
```

    replacementText: z.string().min(1).max(2000),

Rules:
- Read the Claim text, then read each cited slice.
- Decide ONE verdict:
  - "supported" — the cited slices, taken together, contain enough information to substantiate the claim.
  - "unsupported" — no cited slice contains the information; the claim might still be true, but the citations don't show it.
  - "contradicted" — at least one cited slice asserts something incompatible with the claim.
- Quote evidence verbatim from a cited slice in "evidenceText" — do NOT paraphrase.
- For "unsupported" or "contradicted", propose a Correction: { replacementText } that is supported by the cited slices.

Return ONLY JSON: { verdict, evidenceText, correction?: { replacementText, newCitationLabel? } }.`;

      const block = citedSlices.map((s) => `<slice citationId="${s.citationId}">\n${s.sliceText}\n</slice>`).join('\n');
      const { result } = await client.jsonComplete({
        model: VERIFIER_MODEL,
        system: SYSTEM,
        messages: [{ role: 'user', content: `Claim:\n${claim.claimText}\n\nCited slices:\n${block}` }],
        schema: VerdictSchema,
        temperature: 0,
        maxTokens: 800,
      });
      return result;
    },
  };
};

import type { LintEvent } from '@package/contracts/verification';
import { lintWiki } from '../../application/lint-wiki.ts';
import type { LintRuntimeDeps } from '../../application/ports.ts';

interface State { storage: { get<T>(k: string): Promise<T | undefined>; put(k: string, v: unknown): Promise<void> }; waitUntil(p: Promise<unknown>): void }
interface Env { [k: string]: unknown }

interface RunCmd { kind: 'start'; lintRunId: string; wikiId: string }
const TAPE_KEY = 'tape';

export interface LintRunDOFactoryArgs {
  buildDeps: (env: Env, emit: (e: LintEvent) => Promise<void>) => LintRuntimeDeps;
}

export const createLintRunDOClass = ({ buildDeps }: LintRunDOFactoryArgs) =>
  class LintRunDO {
    private subs = new Set<(e: LintEvent) => void>();
    constructor(private state: State, private env: Env) {}

    async fetch(req: Request): Promise<Response> {
      const url = new URL(req.url);
      if (url.pathname === '/start' && req.method === 'POST') {
        const cmd = (await req.json()) as RunCmd;
        this.state.waitUntil(this.run(cmd));
        return new Response('ok', { status: 202 });
      }
      if (url.pathname === '/subscribe' && req.method === 'GET') {
        const stream = new ReadableStream({
          start: (controller) => {
            const send = (e: LintEvent) => controller.enqueue(`data: ${JSON.stringify(e)}\n\n`);
            this.subs.add(send);
            this.state.storage.get<LintEvent[]>(TAPE_KEY).then((tape) => { for (const e of tape ?? []) send(e); });
          },
        });
        return new Response(stream, { headers: { 'content-type': 'text/event-stream' } });
      }
      return new Response('not found', { status: 404 });
    }

    private async run(cmd: RunCmd) {
      const tape: LintEvent[] = [];
      const emit = async (e: LintEvent) => {
        tape.push(e);
        await this.state.storage.put(TAPE_KEY, tape);
        for (const s of this.subs) s(e);
      };
      const deps = buildDeps(this.env, emit);
      try {
        await lintWiki(deps, { lintRunId: cmd.lintRunId as never, wikiId: cmd.wikiId as never });
      } catch (err) {
        await emit({ kind: 'LintRunFailed', lintRunId: cmd.lintRunId as never, message: String(err) });
      }
    }
  };
```
