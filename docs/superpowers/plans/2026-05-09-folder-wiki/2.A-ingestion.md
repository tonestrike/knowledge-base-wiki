**Goal:** Make `Source` real — Drive OAuth, fetch, extract (PDF / Google Doc / Sheet / Slide → text + outline + page images), persist to D1 + R2, expose every procedure declared in `@package/contracts/ingestion`, and emit `SourceIngested` domain events for the wiki context to subscribe to.

**Tech Stack:** `googleapis` 144.x (Drive v3), `pdfjs-dist` 4.10.x, `@anthropic-ai/sdk` not used here, Cloudflare D1 + R2 + KV, `@orpc/server`, `zod`. The `EventBus` interface is defined here in `@package/shared-kernel/events` so that 2.B/2.C/2.D consume the same shape.

// packages/shared-kernel/src/events/index.ts
export interface DomainEventEnvelope<Name extends string = string, Payload = unknown> {
  readonly name: Name;
  readonly occurredAt: string;
  readonly payload: Payload;
}

export type EventHandler<E extends DomainEventEnvelope = DomainEventEnvelope> = (event: E) => void | Promise<void>;

export interface EventBus {
  publish<E extends DomainEventEnvelope>(event: E): Promise<void>;
  subscribe<Name extends string>(name: Name, handler: EventHandler<DomainEventEnvelope<Name>>): () => void;
}

export class InMemoryEventBus implements EventBus {
  private handlers = new Map<string, Set<EventHandler>>();

  subscribe<Name extends string>(name: Name, handler: EventHandler<DomainEventEnvelope<Name>>): () => void {
    const set = this.handlers.get(name) ?? new Set<EventHandler>();
    set.add(handler as EventHandler);
    this.handlers.set(name, set);
    return () => set.delete(handler as EventHandler);
  }

  async publish<E extends DomainEventEnvelope>(event: E): Promise<void> {
    const set = this.handlers.get(event.name);
    if (!set) return;
    for (const h of set) {
      try {
        await h(event);
      } catch (err) {
        // The bus must not let one bad subscriber block the others.
        // Production bindings (e.g. Cloudflare Queue) DLQ on retry; in-memory swallows.
        console.error('[EventBus] subscriber threw', event.name, err);
      }
    }
  }
}
```

import type { ContentHash, FolderId, SourceId } from '@package/contracts/shared';

export interface Manifest {
  readonly driveFileId: string;
  readonly filename: string;
  readonly mime:
    | 'application/pdf'
    | 'application/vnd.google-apps.document'
    | 'application/vnd.google-apps.spreadsheet'
    | 'application/vnd.google-apps.presentation'
    | 'text/plain'
    | 'text/markdown';
  readonly sizeBytes: number;
  readonly modifiedAt: string;
  readonly pageCount?: number;
}

export interface Source {
  readonly id: SourceId;
  readonly folderId: FolderId;
  readonly manifest: Manifest;
  readonly contentHash: ContentHash;
  readonly fetchedAt: string;
}

export const Source = {
  create(props: {
    id: SourceId;
    folderId: FolderId;
    manifest: Manifest;
    contentHash: ContentHash;
    fetchedAt: string;
  }): Source {
    return Object.freeze({ ...props, manifest: Object.freeze({ ...props.manifest }) });
  },
};

export const sourceFingerprint = (s: Source): string =>
  `${s.manifest.driveFileId}:${s.contentHash}`;

import type { ByteRange } from '@package/contracts/shared';

export interface OutlineNode {
  readonly kind: 'heading' | 'table' | 'figure';
  readonly level: number;
  readonly title: string;
  readonly byteRange: ByteRange;
  readonly page?: number;
}

export interface Outline {
  readonly nodes: ReadonlyArray<OutlineNode>;
}

export const Outline = {
  fromNodes(nodes: OutlineNode[]): Outline {
    return Object.freeze({ nodes: Object.freeze([...nodes]) });
  },
  empty(): Outline {
    return Object.freeze({ nodes: Object.freeze([]) });
  },
};

export const outlineDepth = (o: Outline): number =>
  o.nodes.reduce((max, n) => Math.max(max, n.level), 0);

import type { IngestionDeps } from './ports.ts';

export async function startDriveAuth(deps: IngestionDeps): Promise<{ authorizationUrl: string; state: string }> {
  const { drive, oauthState, now } = deps;
  const { state, authorizationUrl } = await drive.startAuth();
  await oauthState.set(state, { createdAt: now().toISOString() });
  return { authorizationUrl, state };
}
```

```ts
// packages/domains/ingestion/src/application/complete-drive-auth.ts
import type { UserId } from '@package/contracts/shared';
import type { IngestionDeps } from './ports.ts';

export async function completeDriveAuth(
  deps: IngestionDeps,
  input: { code: string; state: string },
): Promise<{ userId: UserId; scopes: string[] }> {
  const { drive, oauthState, oauth } = deps;
  const stored = await oauthState.consume(input.state);
  if (!stored) throw new Error('unknown or expired oauth state');
  const tokens = await drive.completeAuth(input);
  await oauth.saveTokens({
    userId: tokens.userId,
    refreshToken: tokens.refreshToken,
    accessToken: tokens.accessToken,
    expiresAt: tokens.expiresAt,
  });
  return { userId: tokens.userId, scopes: ['drive.readonly'] };
}

// packages/domains/ingestion/src/application/extract-source.ts
import type { Manifest } from '../domain/source.ts';
import type { Outline } from '../domain/outline.ts';

export interface Extractor {
  extract(input: { bytes: Uint8Array }): Promise<{
    text: string;
    outline: Outline;
    pageCount: number;
    pageImages: Array<{ pageNumber: number; png: Uint8Array }>;
  }>;
}

export interface ExtractorRegistry {
  pdf: Extractor;
  doc: Extractor;
  sheet: Extractor;
  slide: Extractor;
}

export async function extractSource(
  extractors: ExtractorRegistry,
  input: { mime: Manifest['mime']; bytes: Uint8Array },
) {
  switch (input.mime) {
    case 'application/pdf':
    case 'text/plain':
    case 'text/markdown':
      return extractors.pdf.extract({ bytes: input.bytes });
    case 'application/vnd.google-apps.document':
      return extractors.doc.extract({ bytes: input.bytes });
    case 'application/vnd.google-apps.spreadsheet':
      return extractors.sheet.extract({ bytes: input.bytes });
    case 'application/vnd.google-apps.presentation':
      return extractors.slide.extract({ bytes: input.bytes });
    default: {
      const _exhaustive: never = input.mime;
      throw new Error(`unsupported mime: ${_exhaustive}`);
    }
  }
}

// packages/domains/ingestion/src/application/ingest-folder.ts
import { sourceId, type FolderId, type SourceId } from '@package/contracts/shared';
import { Source } from '../domain/source.ts';
import { extractSource, type ExtractorRegistry } from './extract-source.ts';
import type { IngestionDeps } from './ports.ts';

export interface IngestFolderDeps extends IngestionDeps {
  extractors: ExtractorRegistry;
}

const sha256Hex = async (bytes: Uint8Array): Promise<string> => {
  const buf = await crypto.subtle.digest('SHA-256', bytes);
  return [...new Uint8Array(buf)].map((b) => b.toString(16).padStart(2, '0')).join('');
};

export async function ingestFolder(
  deps: IngestFolderDeps,
  input: { folderId: FolderId; driveFileIds: string[] },
): Promise<{ successful: number; failed: number }> {
  let successful = 0;
  let failed = 0;
  for (const driveFileId of input.driveFileIds) {
    try {
      await ingestOne(deps, input.folderId, driveFileId);
      successful++;
    } catch (err) {
      failed++;
      console.error('[ingestion] failed', driveFileId, err);
    }
  }
  return { successful, failed };
}

async function ingestOne(deps: IngestFolderDeps, folderId: FolderId, driveFileId: string): Promise<void> {
  const fetched = await deps.drive.fetch({ driveFileId });
  const contentHash = `sha256:${await sha256Hex(fetched.bytes)}` as never;

  const existing = await deps.sources.findByDriveFileId({ folderId, driveFileId });
  if (existing && existing.contentHash === contentHash) return; // already ingested at this content hash

  const id = sourceId(deps.newId());
  await deps.storage.putRaw({ sourceId: id, bytes: fetched.bytes, contentHash });

  const extracted = await extractSource(deps.extractors, { mime: fetched.manifest.mime, bytes: fetched.bytes });
  await deps.storage.putText({ sourceId: id, text: extracted.text });
  await deps.storage.putOutline({ sourceId: id, outline: extracted.outline });
  for (const img of extracted.pageImages) {
    await deps.storage.putPageImage({ sourceId: id, pageNumber: img.pageNumber, png: img.png });
  }

  const source = Source.create({
    id,
    folderId,
    manifest: { ...fetched.manifest, pageCount: extracted.pageCount },
    contentHash,
    fetchedAt: deps.now().toISOString(),
  });
  await deps.sources.insert(source);

  await deps.eventBus.publish({
    name: 'SourceIngested',
    occurredAt: deps.now().toISOString(),
    payload: {
      sourceId: source.id,
      folderId: source.folderId,
      contentHash: source.contentHash,
      filename: source.manifest.filename,
    },
  });
}

  drive_file_id: string;

  size_bytes: number;

      filename: r.filename,
      mime: r.mime,

  mime: s.manifest.mime,
  contentHash: s.contentHash,
  pageCount: s.manifest.pageCount,
  fetchedAt: s.fetchedAt,
});

export const createD1SourceRepository = (db: D1Database): SourceRepository => ({
  async insert(s) {
    await db
      .prepare(
        'INSERT INTO sources (id, folder_id, drive_file_id, filename, mime, size_bytes, modified_at, page_count, content_hash, fetched_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)',
      )

        s.id,
        s.folderId,
        s.manifest.driveFileId,
        s.manifest.filename,
        s.manifest.mime,
        s.manifest.sizeBytes,
        s.manifest.modifiedAt,
        s.manifest.pageCount ?? null,
        s.contentHash,
        s.fetchedAt,
      )
      .run();

    return row ? rowToSource(row) : null;

  async findByDriveFileId({ folderId, driveFileId }) {
    const row = await db
      .prepare('SELECT * FROM sources WHERE folder_id = ? AND drive_file_id = ? ORDER BY fetched_at DESC LIMIT 1')
      .bind(folderId, driveFileId)
      .first();
    return row ? rowToSource(row) : null;
  },

  async list({ folderId, cursor, limit }) {
    const cursorClause = cursor ? 'AND fetched_at < ?' : '';
    const rows = await db
      .prepare(
        `SELECT * FROM sources WHERE folder_id = ? ${cursorClause} ORDER BY fetched_at DESC LIMIT ?`,
      )
      .bind(...(cursor ? [folderId, cursor, limit] : [folderId, limit]))
      .all();
    return {
      items: rows.results.map(rowToSource),
      nextCursor: rows.results[rows.results.length - 1]?.fetched_at,
    };
  },

  toWire: sourceToWire,

  async completeAuth({ code }) {
    const tokenRes = await fetch('https://oauth2.googleapis.com/token', {
      method: 'POST',
      headers: { 'content-type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({
        code,
        client_id: cfg.clientId,
        client_secret: cfg.clientSecret,
        redirect_uri: cfg.redirectUri,
        grant_type: 'authorization_code',
      }),

    const t = (await tokenRes.json()) as { access_token: string; refresh_token: string; expires_in: number };
    const profileRes = await fetch('https://www.googleapis.com/oauth2/v2/userinfo', {
      headers: { authorization: `Bearer ${t.access_token}` },
    });
    const profile = (await profileRes.json()) as { email: string };

      userId: cfg.resolveUserId(profile.email),
      refreshToken: t.refresh_token,
      accessToken: t.access_token,

  async listFolders({ parentId, query, limit }) {
    const accessToken = await getCurrentAccessToken();
    const q = [
      "mimeType = 'application/vnd.google-apps.folder'",
      'trashed = false',
      parentId ? `'${parentId}' in parents` : '',
      query ? `name contains '${query.replaceAll("'", "\\'")}'` : '',
    ].filter(Boolean).join(' and ');
    const u = new URL('https://www.googleapis.com/drive/v3/files');
    u.searchParams.set('q', q);
    u.searchParams.set('fields', 'files(id,name,modifiedTime),nextPageToken');
    u.searchParams.set('pageSize', String(limit));
    const res = await fetch(u, { headers: { authorization: `Bearer ${accessToken}` } });
    if (!res.ok) throw new Error(`drive listFolders ${res.status}`);
    const j = (await res.json()) as { files: Array<{ id: string; name: string; modifiedTime: string }>; nextPageToken?: string };
    return {
      folders: j.files.map((f) => ({ driveFolderId: f.id, name: f.name, modifiedAt: f.modifiedTime })),
      nextPageToken: j.nextPageToken,
    };
  },

  async fetch({ driveFileId }) {
    const accessToken = await getCurrentAccessToken();
    const meta = await fetch(`https://www.googleapis.com/drive/v3/files/${driveFileId}?fields=id,name,mimeType,size,modifiedTime`, {
      headers: { authorization: `Bearer ${accessToken}` },
    }).then((r) => r.json() as Promise<{ name: string; mimeType: string; size?: string; modifiedTime: string }>);
    const isGoogleNative = meta.mimeType.startsWith('application/vnd.google-apps.');
    const exportMime = isGoogleNative
      ? meta.mimeType === 'application/vnd.google-apps.document' ? 'text/plain'
        : meta.mimeType === 'application/vnd.google-apps.spreadsheet' ? 'text/csv'
        : meta.mimeType === 'application/vnd.google-apps.presentation' ? 'application/pdf'
        : 'application/pdf'
      : null;
    const dl = exportMime
      ? `https://www.googleapis.com/drive/v3/files/${driveFileId}/export?mimeType=${encodeURIComponent(exportMime)}`
      : `https://www.googleapis.com/drive/v3/files/${driveFileId}?alt=media`;
    const res = await fetch(dl, { headers: { authorization: `Bearer ${accessToken}` } });
    if (!res.ok) throw new Error(`drive fetch ${res.status}`);
    const bytes = new Uint8Array(await res.arrayBuffer());
    return {
      manifest: {
        driveFileId,
        filename: meta.name,
        mime: (isGoogleNative ? meta.mimeType : meta.mimeType) as 'application/pdf',
        sizeBytes: meta.size ? Number(meta.size) : bytes.length,
        modifiedAt: meta.modifiedTime,
      },
      bytes,
    };
  },
});

// packages/domains/ingestion/src/infrastructure/pdf-extractor.ts
import { getDocument } from 'pdfjs-dist/legacy/build/pdf.mjs';
import type { Extractor } from '../application/extract-source.ts';
import { Outline } from '../domain/outline.ts';

export const createPdfExtractor = (): Extractor => ({
  async extract({ bytes }) {
    const doc = await getDocument({ data: bytes }).promise;
    const pageCount = doc.numPages;
    const nodes: Array<{ kind: 'heading'; level: number; title: string; byteRange: { start: number; end: number }; page: number }> = [];
    let cursor = 0;
    let full = '';
    const pageImages: Array<{ pageNumber: number; png: Uint8Array }> = [];

    for (let p = 1; p <= pageCount; p++) {
      const page = await doc.getPage(p);
      const content = await page.getTextContent();
      for (const item of content.items) {
        if (!('str' in item)) continue;
        const t = item.str;
        if (!t) continue;
        const start = cursor;
        const end = start + t.length;
        // crude heading heuristic: short, all-caps or title-case lines
        if (t.length < 80 && /^[A-Z]/.test(t.trim()) && t.trim() !== t.toLowerCase()) {
          nodes.push({ kind: 'heading', level: 2, title: t.trim(), byteRange: { start, end }, page: p });
        }
        full += t + ' ';
        cursor = end + 1;
      }
      full += '\n';
      cursor += 1;

      // Page-image rendering is skipped in this slice — Phase 2.E renders client-side via PDF.js.
      // If we need server-rendered thumbnails later, add `node-canvas` and render here.
    }

    return {
      text: full,
      outline: Outline.fromNodes(nodes),
      pageCount,
      pageImages,
    };
  },
});

const createAesGcmCipher = (keyBase64: string) => {
  let cached: CryptoKey | null = null;
  const key = async () =>
    cached ??
    (cached = await crypto.subtle.importKey(
      'raw',
      Uint8Array.from(atob(keyBase64), (c) => c.charCodeAt(0)),
      { name: 'AES-GCM' },
      false,
      ['encrypt', 'decrypt'],
    ));
  const enc = new TextEncoder();
  const dec = new TextDecoder();
  return {
    async encrypt(plaintext: string) {
      const iv = crypto.getRandomValues(new Uint8Array(12));
      const ct = new Uint8Array(await crypto.subtle.encrypt({ name: 'AES-GCM', iv }, await key(), enc.encode(plaintext)));
      const out = new Uint8Array(iv.length + ct.length);
      out.set(iv); out.set(ct, iv.length);
      return btoa(String.fromCharCode(...out));
    },
    async decrypt(ciphertext: string) {
      const all = Uint8Array.from(atob(ciphertext), (c) => c.charCodeAt(0));
      const iv = all.slice(0, 12);
      const ct = all.slice(12);
      const pt = await crypto.subtle.decrypt({ name: 'AES-GCM', iv }, await key(), ct);
      return dec.decode(pt);
    },
  };
};

app.use('/rpc/*', cors({ origin: (o) => o, credentials: true }));

const handler = new RPCHandler(router, {
  interceptors: [onError((error) => console.error('[orpc]', error))],
});

app.use('/rpc/*', async (c, next) => {
  const ingestion = buildIngestion(c.env);
  const { matched, response } = await handler.handle(c.req.raw, {
    prefix: '/rpc',
    context: {
      clock: systemClock,
      ...ingestion, // spreads into all routers; routers ignore deps they don't need
    },
  });
  if (matched) return c.newResponse(response.body, response);
  await next();
});

app.get('/', (c) => c.text('tenex api'));

export default app;
```
