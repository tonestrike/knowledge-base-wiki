# Slice 2.B — wiki (Compile pipeline + adaptive schema)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. This is the dominant slice in the Phase-2 fan-out — sequence carefully.

**Goal:** Make compile real. SchemaInferrer (Sonnet 4.6) reads the first 5–10 Sources, declares a `WikiSchema`. Planner decomposes the run; Researcher (×N, Haiku 4.5) gathers per-Source findings; Drafter (Sonnet 4.6) composes typed `ConceptPage` and `SummaryPage` markdown; Linker (Haiku) resolves Backlinks via the schema's Relations; IndexBuilder (Haiku) emits one `IndexPage` per `PageType`. A `CompileRunDO` orchestrates the multi-minute run, persists state across Worker invocations, and fans out SSE events to every subscriber. Subscribes to `SourceIngested` for incremental compile (wired but driven manually in v1). Emits `CompileFinished` for the verification context to lint.

**Architecture:** Same DDD shape as 2.A. `domain/` carries pure aggregates and value objects (`Wiki`, `WikiPage`, `WikiSchema`, `PageType`, `Relation`, `Backlink`, `Claim`, `Citation`, `CompileRun`, `CompileEvent`). `application/` has framework-free use-cases (`infer-schema`, `plan-compile`, `research-source`, `draft-page`, `resolve-backlinks`, `build-indexes`, `compile-folder`, `get-wiki`, `get-schema`, `get-page`, `list-wikis`, `list-pages`, `get-compile-run`). `infrastructure/` wraps Anthropic SDK, D1, R2, and the new `CompileRunDO` (`durable_objects/compile-run-do.ts`). The DO holds the run's transient state in its own `state.storage` and exposes a `subscribe(controller)` shape so the oRPC `streamCompileEvents` handler can pull events with simple `for await ... of`.

**Tech Stack:** `@anthropic-ai/sdk` 0.40.x, Cloudflare Durable Objects (Workers Paid Plan required), D1, R2. `@orpc/server` event iterator wraps the DO subscription.

export interface Wiki {
  readonly id: WikiId;
  readonly folderId: FolderId;
  readonly schema: WikiSchema;
  readonly createdAt: string;
  readonly updatedAt: string;
  readonly lastCompiledAt?: string;
  readonly pageCount: number;

export const Wiki = {
  create(props: {
    id: WikiId;
    folderId: FolderId;
    schema: WikiSchema;
    createdAt: string;
  }): Wiki {
    if (props.schema.pageTypes.length === 0) {
      throw new Error('Wiki requires at least one PageType in its schema');
    }
    return Object.freeze({
      ...props,
      updatedAt: props.createdAt,
      pageCount: 0,
    });
  },
  recordCompile(wiki: Wiki, at: string, pageCount: number): Wiki {
    return Object.freeze({ ...wiki, updatedAt: at, lastCompiledAt: at, pageCount });
  },
};

// packages/domains/wiki/src/domain/wiki-page.ts
import type {
  Citation,
  Claim,
  SourceId,
  WikiId,
  WikiPageId,
} from '@package/contracts/shared';
import type { Backlink } from './backlink.ts';

export type WikiPageSubtype = 'Concept' | 'Summary' | 'Answer' | 'Index';

interface BasePage {
  readonly id: WikiPageId;
  readonly wikiId: WikiId;
  readonly subtype: WikiPageSubtype;
  readonly slug: string;
  readonly title: string;
  readonly body: string;
  readonly claims: ReadonlyArray<Claim>;
  readonly citations: ReadonlyArray<Citation>;
  readonly backlinks: ReadonlyArray<Backlink>;
  readonly updatedAt: string;
  readonly pageType?: string;
}

export interface ConceptPage extends BasePage {
  readonly subtype: 'Concept';
  readonly pageType: string;
}

export interface SummaryPage extends BasePage {
  readonly subtype: 'Summary';
  readonly sourceId: SourceId;
}

export interface AnswerPage extends BasePage {
  readonly subtype: 'Answer';
  readonly question: string;
}

export interface IndexEntry {
  readonly pageId: WikiPageId;
  readonly title: string;
  readonly summary?: string;
}

export interface IndexPage extends BasePage {
  readonly subtype: 'Index';
  readonly pageType: string;
  readonly entries: ReadonlyArray<IndexEntry>;
}

export type WikiPage = ConceptPage | SummaryPage | AnswerPage | IndexPage;

const collectCitations = (claims: ReadonlyArray<Claim>): ReadonlyArray<Citation> => {
  const seen = new Set<string>();
  const out: Citation[] = [];
  for (const c of claims) for (const cite of c.citations) {
    if (seen.has(cite.id)) continue;
    seen.add(cite.id);
    out.push(cite);
  }
  return out;
};

export const WikiPage = {
  concept(props: Omit<ConceptPage, 'subtype' | 'citations' | 'backlinks'> & { backlinks?: Backlink[] }): ConceptPage {
    if (!props.pageType) throw new Error('ConceptPage requires a pageType');
    return Object.freeze({
      ...props,
      subtype: 'Concept' as const,
      citations: collectCitations(props.claims),
      backlinks: Object.freeze([...(props.backlinks ?? [])]),
      claims: Object.freeze([...props.claims]),
    });
  },
  summary(props: Omit<SummaryPage, 'subtype' | 'citations' | 'backlinks'> & { backlinks?: Backlink[] }): SummaryPage {
    return Object.freeze({
      ...props,
      subtype: 'Summary' as const,
      citations: collectCitations(props.claims),
      backlinks: Object.freeze([...(props.backlinks ?? [])]),
      claims: Object.freeze([...props.claims]),
    });
  },
  answer(props: Omit<AnswerPage, 'subtype' | 'citations' | 'backlinks'> & { backlinks?: Backlink[] }): AnswerPage {
    return Object.freeze({
      ...props,
      subtype: 'Answer' as const,
      citations: collectCitations(props.claims),
      backlinks: Object.freeze([...(props.backlinks ?? [])]),
      claims: Object.freeze([...props.claims]),
    });
  },
  index(props: Omit<IndexPage, 'subtype' | 'citations' | 'claims' | 'body' | 'backlinks'> & { backlinks?: Backlink[] }): IndexPage {
    if (props.entries.length === 0) throw new Error('IndexPage requires at least one entry');
    const body = props.entries.map((e) => `- [${e.title}](/${e.pageId})${e.summary ? ` — ${e.summary}` : ''}`).join('\n');
    return Object.freeze({
      ...props,
      subtype: 'Index' as const,
      claims: Object.freeze([]),
      citations: Object.freeze([]),
      backlinks: Object.freeze([...(props.backlinks ?? [])]),
      body,
    });
  },
};
```

// packages/domains/wiki/src/domain/backlink.ts
import type { Cardinality, Relation, WikiPageId } from '@package/contracts/shared';

export interface Backlink {
  readonly fromPageId: WikiPageId;
  readonly toPageId: WikiPageId;
  readonly relationName?: string;
}

export const Backlink = {
  create(props: { fromPageId: WikiPageId; toPageId: WikiPageId; relationName?: string }): Backlink {
    if (props.fromPageId === props.toPageId) {
      throw new Error('Backlink cannot point to its own page (self-link forbidden)');
    }
    return Object.freeze({ ...props });
  },
};

export const validateRelationArity = (
  backlinks: ReadonlyArray<Backlink>,
  relations: ReadonlyArray<Relation>,
): string[] => {
  const errors: string[] = [];
  const byRelation = new Map<string, Backlink[]>();
  for (const bl of backlinks) {
    if (!bl.relationName) continue;
    byRelation.set(bl.relationName, [...(byRelation.get(bl.relationName) ?? []), bl]);
  }
  const cardinalityOf = new Map<string, Cardinality>(relations.map((r) => [r.name, r.cardinality]));
  for (const [rel, bls] of byRelation) {
    const card = cardinalityOf.get(rel);
    if (!card) continue;
    if (card === 'one-to-one' || card === 'many-to-one') {
      const seen = new Set<string>();
      for (const bl of bls) {
        if (seen.has(bl.fromPageId)) {
          errors.push(`relation "${rel}" cardinality ${card} violated by ${bl.fromPageId}`);
        }
        seen.add(bl.fromPageId);
      }
    }
  }
  return errors;
};
```

// packages/domains/wiki/src/domain/compile-run.ts
import type { CompileRunId, FolderId, WikiId } from '@package/contracts/shared';

export type CompileRunStatus =
  | 'pending'
  | 'inferring-schema'
  | 'planning'
  | 'researching'
  | 'drafting'
  | 'linking'
  | 'indexing'
  | 'finished'
  | 'failed';

const ORDER: CompileRunStatus[] = [
  'pending', 'inferring-schema', 'planning', 'researching',
  'drafting', 'linking', 'indexing', 'finished',
];

export interface CompileRun {
  readonly id: CompileRunId;
  readonly folderId: FolderId;
  readonly wikiId?: WikiId;
  readonly status: CompileRunStatus;
  readonly startedAt: string;
  readonly schemaInferredAt?: string;
  readonly endedAt?: string;
  readonly failureMessage?: string;
}

export const CompileRun = {
  start(props: { id: CompileRunId; folderId: FolderId; startedAt: string }): CompileRun {
    return Object.freeze({ ...props, status: 'pending' as const });
  },
  advance(run: CompileRun, next: CompileRunStatus, at: string): CompileRun {
    const i = ORDER.indexOf(run.status);
    const j = ORDER.indexOf(next);
    if (j !== i + 1 || j === -1) throw new Error(`illegal transition ${run.status} → ${next}`);
    const patch = next === 'planning' && run.schemaInferredAt === undefined
      ? { schemaInferredAt: at }
      : {};
    return Object.freeze({ ...run, ...patch, status: next });
  },
  finish(run: CompileRun, props: { wikiId: WikiId; endedAt: string }): CompileRun {
    if (run.status !== 'indexing') throw new Error(`cannot finish from ${run.status}`);
    return Object.freeze({ ...run, status: 'finished', wikiId: props.wikiId, endedAt: props.endedAt });
  },
  fail(run: CompileRun, message: string, at: string): CompileRun {
    return Object.freeze({ ...run, status: 'failed', failureMessage: message, endedAt: at });
  },
};
```

// packages/domains/wiki/src/application/draft-page.ts
import { z } from 'zod';
import { sourceId as parseSourceId, type ContentHash, type SourceId } from '@package/contracts/shared';
import type { AnthropicClient } from './ports.ts';

export const DRAFTER_MODEL = 'claude-sonnet-4-6';

const DraftOutput = z.object({
  title: z.string().min(1).max(200),
  slug: z.string().min(1).max(120).regex(/^[a-z0-9-]+$/),
  body: z.string().min(1),
  claims: z.array(
    z.object({
      paragraphId: z.string().min(1),
      claimText: z.string().min(1).max(2000),
      citations: z.array(
        z.object({
          sourceId: z.string().uuid(),
          spanStart: z.number().int().nonnegative(),
          spanEnd: z.number().int().positive(),
          label: z.string().min(1).max(200),
        }),
      ).min(1),
    }),
  ).min(1),
});

const SYSTEM = `You are Drafter. Compose a magazine-quality wiki page in markdown.

Rules:
- Title: a short noun phrase ("Decision: Expand into EMEA in Q4"), not a sentence.
- Body: 3-6 short sections with H2 headings; serif-prose register.
- Each meaningful claim in the body must be paired with a citation.
- Each citation MUST point to one of the provided findings by (sourceId, spanStart, spanEnd) — DO NOT invent byte ranges.
- The "label" is what the reader sees on the chip (e.g. "Q3 minutes, p.4"). Make it specific.

Return ONLY JSON: { title, slug, body, claims: [{ paragraphId, claimText, citations: [{ sourceId, spanStart, spanEnd, label }] }] }.`;

export interface ResearchFinding {
  sourceId: SourceId;
  sourceFilename: string;
  sourceText: string;
  sourceContentHash: ContentHash;
  evidence: string;
  spanStart: number;
  spanEnd: number;
  title: string;
}

export async function draftPage(
  deps: { anthropic: AnthropicClient },
  input: { pageType: string; findings: ResearchFinding[] },
) {
  const findingsBlock = input.findings.map((f) =>
    `<finding sourceId="${f.sourceId}" filename="${f.sourceFilename}" spanStart="${f.spanStart}" spanEnd="${f.spanEnd}">\n${f.evidence}\n</finding>`,
  ).join('\n');

  const { result, inputTokens, outputTokens } = await deps.anthropic.jsonComplete({
    model: DRAFTER_MODEL,
    system: SYSTEM,
    messages: [{ role: 'user', content: `PageType: ${input.pageType}\n\nFindings:\n${findingsBlock}` }],
    schema: DraftOutput,
    maxTokens: 3000,
    temperature: 0.3,
  });

  // Map citation IDs back to source content hashes (the Drafter does not invent these).
  const byId = new Map(input.findings.map((f) => [f.sourceId, f]));
  const enriched = result.claims.map((c) => ({
    ...c,
    citations: c.citations.map((cite) => {
      const f = byId.get(parseSourceId(cite.sourceId) as SourceId);
      if (!f) throw new Error(`Drafter cited unknown sourceId: ${cite.sourceId}`);
      // Snap the byte range to the exact finding range — defends against
      // model arithmetic drift inside the citation.
      return { ...cite, sourceContentHash: f.sourceContentHash };
    }),
  }));

  return { draft: { ...result, claims: enriched }, usage: { inputTokens, outputTokens } };
}

// packages/domains/wiki/src/application/resolve-backlinks.ts
import type { Relation, WikiPageId } from '@package/contracts/shared';
import { wikiPageId as parsePageId } from '@package/contracts/shared';
import { Backlink, validateRelationArity } from '../domain/backlink.ts';
import type { WikiPage } from '../domain/wiki-page.ts';

const LINK_RE = /\]\(\/([0-9a-f-]{36})\)/g;

export function resolveBacklinks(
  pages: ReadonlyArray<WikiPage>,
  relations: ReadonlyArray<Relation>,
): { backlinks: Backlink[]; arityErrors: string[] } {
  const knownIds = new Set(pages.map((p) => p.id));
  const byPageType = new Map(pages.map((p) => [p.id, (p as { pageType?: string }).pageType ?? '']));
  const out: Backlink[] = [];

  for (const page of pages) {
    LINK_RE.lastIndex = 0;
    const seen = new Set<string>();
    for (const m of page.body.matchAll(LINK_RE)) {
      const targetIdRaw = m[1];
      if (!targetIdRaw) continue;
      const targetId = parsePageId(targetIdRaw) as WikiPageId;
      if (!knownIds.has(targetId) || targetId === page.id || seen.has(targetIdRaw)) continue;
      seen.add(targetIdRaw);

      const fromType = byPageType.get(page.id) ?? '';
      const toType = byPageType.get(targetId) ?? '';
      const relationName = relations.find((r) => r.from === fromType && r.to === toType)?.name;

      out.push(Backlink.create({ fromPageId: page.id, toPageId: targetId, relationName }));
    }
  }

  const arityErrors = validateRelationArity(out, relations);
  return { backlinks: out, arityErrors };
}

// packages/domains/wiki/src/application/build-indexes.ts
import { wikiPageId, type WikiId } from '@package/contracts/shared';
import { WikiPage, type ConceptPage, type IndexPage } from '../domain/wiki-page.ts';

export function buildIndexes(input: {
  wikiId: WikiId;
  pages: ReadonlyArray<WikiPage | ConceptPage>;
  pageTypes: ReadonlyArray<string>;
  newId: () => string;
  now: () => Date;
}): { indexPages: IndexPage[] } {
  const concepts = input.pages.filter((p): p is ConceptPage => p.subtype === 'Concept');
  const out: IndexPage[] = [];

  for (const pt of input.pageTypes) {
    const matching = concepts.filter((c) => c.pageType === pt);
    if (matching.length === 0) continue;
    out.push(
      WikiPage.index({
        id: wikiPageId(input.newId()),
        wikiId: input.wikiId,
        pageType: pt,
        slug: `index-${pt.toLowerCase()}`,
        title: pluralize(pt),
        entries: matching.map((c) => ({ pageId: c.id, title: c.title })),
        updatedAt: input.now().toISOString(),
      }),
    );
  }
  return { indexPages: out };
}

const pluralize = (s: string): string => (s.endsWith('s') ? s : `${s}s`);

// packages/domains/wiki/src/application/compile-folder.ts
import { wikiId, wikiPageId, type CompileRunId, type FolderId, type WikiId, type WikiPageId } from '@package/contracts/shared';
import { CompileRun } from '../domain/compile-run.ts';
import { Wiki } from '../domain/wiki.ts';
import { WikiPage, type WikiPage as WP } from '../domain/wiki-page.ts';
import { buildIndexes } from './build-indexes.ts';
import { draftPage, type ResearchFinding } from './draft-page.ts';
import { inferSchema } from './infer-schema.ts';
import { planCompile } from './plan-compile.ts';
import { researchSource } from './research-source.ts';
import { resolveBacklinks } from './resolve-backlinks.ts';
import type { CompileRuntimeDeps } from './ports.ts';

export async function compileFolder(
  deps: CompileRuntimeDeps,
  input: { compileRunId: CompileRunId; folderId: FolderId },
): Promise<{ wikiId: WikiId }> {
  const startedAt = deps.now().toISOString();
  let run = CompileRun.start({ id: input.compileRunId, folderId: input.folderId, startedAt });
  await deps.runs.insert(run);

  const sourceList = await deps.sources.list(input.folderId);
  if (sourceList.length === 0) throw new Error(`Folder ${input.folderId} has no sources to compile`);

  await deps.emit({ kind: 'CompileStarted', compileRunId: input.compileRunId, folderId: input.folderId, sourceCount: sourceList.length });

  // 1. Schema inference
  run = CompileRun.advance(run, 'inferring-schema', deps.now().toISOString());
  await deps.runs.update(run);
  const sourceTexts = await Promise.all(
    sourceList.slice(0, 10).map(async (s) => {
      const r = await deps.sources.read(s.sourceId);
      if (!r) throw new Error(`Source not readable: ${s.sourceId}`);
      return r;
    }),
  );
  const { schema, reason } = await inferSchema({ anthropic: deps.anthropic }, { sources: sourceTexts });
  await deps.emit({ kind: 'SchemaInferred', compileRunId: input.compileRunId, schema, reason });

  // 2. Wiki record
  const wid = wikiId(deps.newId()) as WikiId;
  const wiki = Wiki.create({ id: wid, folderId: input.folderId, schema, createdAt: deps.now().toISOString() });
  await deps.wikis.insert(wiki);

  // 3. Plan
  run = CompileRun.advance(run, 'planning', deps.now().toISOString());
  await deps.runs.update(run);
  const allTexts = sourceTexts.length === sourceList.length ? sourceTexts : await Promise.all(sourceList.map(async (s) => (await deps.sources.read(s.sourceId))!));
  const { tasks } = await planCompile({ anthropic: deps.anthropic }, { schema, sources: allTexts });

  // 4. Research (concurrent within a cap)
  run = CompileRun.advance(run, 'researching', deps.now().toISOString());
  await deps.runs.update(run);
  const findingsBySource = await Promise.all(
    tasks.map(async (t) => {
      const src = allTexts.find((s) => s.sourceId === t.sourceId);
      if (!src) return [];
      const r = await researchSource({ anthropic: deps.anthropic }, { source: src, pageTypes: t.pageTypes });
      return r.findings.map((f) => ({ ...f, sourceId: src.sourceId, sourceFilename: src.filename, sourceText: src.text }));
    }),
  );
  const allFindings = findingsBySource.flat();

  // 5. Draft per (PageType, source) bucket
  run = CompileRun.advance(run, 'drafting', deps.now().toISOString());
  await deps.runs.update(run);

  const conceptDrafts: WP[] = [];
  for (const pt of schema.pageTypes) {
    const findingsOfType = allFindings.filter((f) => f.pageType === pt.name);
    if (findingsOfType.length === 0) continue;
    const groupedByTitle = groupBy(findingsOfType, (f) => f.title);
    for (const [, findings] of groupedByTitle) {
      const enriched: ResearchFinding[] = await Promise.all(findings.map(async (f) => ({
        sourceId: f.sourceId,
        sourceFilename: f.sourceFilename,
        sourceText: f.sourceText,
        sourceContentHash: ('sha256:' + (await sha256Hex(new TextEncoder().encode(f.sourceText)))) as never,
        evidence: f.evidence,
        spanStart: f.spanStart,
        spanEnd: f.spanEnd,
        title: f.title,
      })));
      const { draft } = await draftPage({ anthropic: deps.anthropic }, { pageType: pt.name, findings: enriched });
      const pid = wikiPageId(deps.newId()) as WikiPageId;
      conceptDrafts.push(WikiPage.concept({
        id: pid, wikiId: wid, pageType: pt.name, slug: draft.slug, title: draft.title, body: draft.body,
        claims: draft.claims.map((c) => ({
          id: '00000000-0000-4000-8000-000000000000' as never, // claims get fresh IDs in the page repo write path
          wikiPageId: pid,
          paragraphId: c.paragraphId,
          claimText: c.claimText,
          citations: c.citations.map((cit, i) => ({
            id: `${pid.slice(0, 8)}-${i.toString().padStart(2, '0')}-${pid.slice(9)}` as never,
            label: cit.label,
            span: { sourceId: cit.sourceId as never, byteRange: { start: cit.spanStart, end: cit.spanEnd }, contentHash: 'sha256:placeholder' as never },
          })),
        })),
        updatedAt: deps.now().toISOString(),
      }));
      await deps.emit({ kind: 'PageDrafted', compileRunId: input.compileRunId, pageId: pid, subtype: 'Concept', pageType: pt.name, title: draft.title });
    }
  }

  // 6. Link
  run = CompileRun.advance(run, 'linking', deps.now().toISOString());
  await deps.runs.update(run);
  const { backlinks } = resolveBacklinks(conceptDrafts, schema.relations);
  for (const bl of backlinks) {
    await deps.emit({ kind: 'BacklinkResolved', compileRunId: input.compileRunId, fromPageId: bl.fromPageId, toPageId: bl.toPageId, relationName: bl.relationName });
  }

  // 7. Index
  run = CompileRun.advance(run, 'indexing', deps.now().toISOString());
  await deps.runs.update(run);
  const { indexPages } = buildIndexes({ wikiId: wid, pages: conceptDrafts, pageTypes: schema.pageTypes.map((p) => p.name), newId: deps.newId, now: deps.now });
  for (const ip of indexPages) {
    await deps.emit({ kind: 'IndexBuilt', compileRunId: input.compileRunId, pageType: ip.pageType, pageCount: ip.entries.length });
  }

  await deps.pages.insertMany([...conceptDrafts, ...indexPages]);

  // 8. Finalize
  const endedAt = deps.now().toISOString();
  run = CompileRun.finish(run, { wikiId: wid, endedAt });
  await deps.runs.update(run);
  const finalWiki = Wiki.recordCompile(wiki, endedAt, conceptDrafts.length + indexPages.length);
  await deps.wikis.update(finalWiki);

  await deps.emit({ kind: 'CompileFinished', compileRunId: input.compileRunId, wikiId: wid, pageCount: conceptDrafts.length + indexPages.length });

  await deps.eventBus.publish({
    name: 'CompileFinished',
    occurredAt: endedAt,
    payload: { compileRunId: input.compileRunId, wikiId: wid, pageCount: conceptDrafts.length + indexPages.length },
  });

  return { wikiId: wid };
}

const groupBy = <T, K>(items: T[], by: (t: T) => K): Map<K, T[]> => {
  const m = new Map<K, T[]>();
  for (const i of items) {
    const k = by(i);
    m.set(k, [...(m.get(k) ?? []), i]);
  }
  return m;
};

const sha256Hex = async (b: Uint8Array): Promise<string> => {
  const buf = await crypto.subtle.digest('SHA-256', b);
  return [...new Uint8Array(buf)].map((x) => x.toString(16).padStart(2, '0')).join('');
};

import Anthropic from '@anthropic-ai/sdk';
import { z } from 'zod';
import type { AnthropicClient } from '../application/ports.ts';

export const createAnthropicClient = (apiKey: string): AnthropicClient => {
  const sdk = new Anthropic({ apiKey });
  return {
    async jsonComplete({ model, system, messages, schema, maxTokens = 2000, temperature = 0 }) {
      const res = await sdk.messages.create({ model, system, messages, max_tokens: maxTokens, temperature });
      const text = res.content.map((b) => (b.type === 'text' ? b.text : '')).join('');
      const json = extractJson(text);
      const parsed = schema.parse(json);
      return { result: parsed, inputTokens: res.usage.input_tokens, outputTokens: res.usage.output_tokens };
    },
  };
};

const extractJson = (s: string): unknown => {
  const fenced = s.match(/```json\s*([\s\S]+?)\s*```/);
  const body = fenced?.[1] ?? s.slice(s.indexOf('{'), s.lastIndexOf('}') + 1);
  return JSON.parse(body);
};

export const _internals = { extractJson };

// packages/domains/wiki/src/infrastructure/durable_objects/compile-run-do.ts
import type { CompileEvent } from '@package/contracts/wiki';
import { compileFolder } from '../../application/compile-folder.ts';
import type { CompileRuntimeDeps } from '../../application/ports.ts';

interface State { storage: { get<T>(k: string): Promise<T | undefined>; put(k: string, v: unknown): Promise<void> }; waitUntil(p: Promise<unknown>): void }

interface Env { /* all bindings the DO needs (DB, STORAGE, ANTHROPIC_API_KEY, etc.) */ [k: string]: unknown }

interface RunCommand {
  kind: 'start';
  compileRunId: string;
  folderId: string;
}

const TAPE_KEY = 'tape';

export interface CompileRunDOFactoryArgs {
  buildDeps: (env: Env, emit: (e: CompileEvent) => Promise<void>) => CompileRuntimeDeps;
}

export const createCompileRunDOClass = ({ buildDeps }: CompileRunDOFactoryArgs) => {
  return class CompileRunDO {
    private subscribers = new Set<(e: CompileEvent) => void>();

    constructor(private readonly state: State, private readonly env: Env) {}

    async fetch(req: Request): Promise<Response> {
      const url = new URL(req.url);
      if (url.pathname === '/start' && req.method === 'POST') {
        const cmd = (await req.json()) as RunCommand;
        this.state.waitUntil(this.run(cmd));
        return new Response('ok', { status: 202 });
      }
      if (url.pathname === '/subscribe' && req.method === 'GET') {
        const stream = new ReadableStream({
          start: (controller) => {
            const send = (e: CompileEvent) => controller.enqueue(`data: ${JSON.stringify(e)}\n\n`);
            this.subscribers.add(send);
            // Replay tape
            this.state.storage.get<CompileEvent[]>(TAPE_KEY).then((tape) => {
              for (const e of tape ?? []) send(e);
            });
          },
        });
        return new Response(stream, { headers: { 'content-type': 'text/event-stream' } });
      }
      return new Response('not found', { status: 404 });
    }

    private async run(cmd: RunCommand) {
      const tape: CompileEvent[] = (await this.state.storage.get<CompileEvent[]>(TAPE_KEY)) ?? [];
      const emit = async (e: CompileEvent) => {
        tape.push(e);
        await this.state.storage.put(TAPE_KEY, tape);
        for (const s of this.subscribers) s(e);
      };
      const deps = buildDeps(this.env, emit);
      try {
        await compileFolder(deps, { compileRunId: cmd.compileRunId as never, folderId: cmd.folderId as never });
      } catch (err) {
        await emit({ kind: 'CompileFailed', compileRunId: cmd.compileRunId as never, message: String(err) });
      }
    }
  };
};
```

import type { CompileEvent } from '@package/contracts/wiki';
import type { CompileRunDispatcher } from '../application/ports.ts';

interface DurableObjectNamespace { idFromName(n: string): { toString(): string }; get(id: { toString(): string }): { fetch(req: Request): Promise<Response> } }

export const createCfCompileRunDispatcher = (ns: DurableObjectNamespace): CompileRunDispatcher => ({
  async start({ compileRunId, folderId }) {
    const stub = ns.get(ns.idFromName(compileRunId));
    await stub.fetch(new Request('https://do/start', { method: 'POST', body: JSON.stringify({ kind: 'start', compileRunId, folderId }) }));
  },
  async *subscribe(compileRunId) {
    const stub = ns.get(ns.idFromName(compileRunId));
    const res = await stub.fetch(new Request('https://do/subscribe'));
    if (!res.body) return;
    const reader = res.body.getReader();
    const decoder = new TextDecoder();
    let buf = '';
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buf += decoder.decode(value, { stream: true });
      let idx;
      while ((idx = buf.indexOf('\n\n')) !== -1) {
        const evt = buf.slice(0, idx).replace(/^data:\s*/m, '').trim();
        buf = buf.slice(idx + 2);
        if (evt) yield JSON.parse(evt) as CompileEvent;
      }
    }
  },
});
